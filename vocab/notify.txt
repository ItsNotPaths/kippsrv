# Notifications, as org.freedesktop.Notifications delivers them. kippbus owns
# the name, so this is the daemon rather than a reader of one.
notif       # <id> app= summary= body= icon= urgency=low|normal|critical read=0|1
            # action= is the spec's action key. exec= is a command the sender
            # attached as the x-kipp-exec hint, and is what a panel runs.
            # <id> ... action= exec=

# Commands an adapter of this domain takes. Uppercase, per SPEC.md. kippbus
# notify takes both on stdin. READ is ours, because the spec has no idea of a
# notification that waits to be looked at.
NOTIFY-CLOSE  # <id>
NOTIFY-READ   # no subject. Everything live becomes read=1
